### **SQL Queries Used:**
```sql
-- Main query for antihypertensive class utilization
WITH age_filter AS (
  SELECT person_id, 
         DATEDIFF(year, DATEFROMPARTS(year_of_birth, ISNULL(month_of_birth, 1), ISNULL(day_of_birth, 1)), GETDATE()) AS age_years
  FROM omop.person
  WHERE DATEDIFF(year, DATEFROMPARTS(year_of_birth, ISNULL(month_of_birth, 1), ISNULL(day_of_birth, 1)), GETDATE()) BETWEEN 40 AND 60
),
antihypertensive_classes AS (
  SELECT concept_id, concept_name, concept_code,
         CASE 
           WHEN concept_code LIKE 'C03%' THEN 'Diuretics'
           WHEN concept_code LIKE 'C07%' THEN 'Beta Blockers'
           WHEN concept_code LIKE 'C08%' THEN 'Calcium Channel Blockers'
           WHEN concept_code LIKE 'C09A%' OR concept_code LIKE 'C09B%' THEN 'ACE Inhibitors'
           WHEN concept_code LIKE 'C09C%' OR concept_code LIKE 'C09D%' THEN 'ARBs'
         END AS drug_class
  FROM omop.concept 
  WHERE vocabulary_id = 'ATC' 
    AND (concept_code LIKE 'C03%' OR concept_code LIKE 'C07%' OR concept_code LIKE 'C08%' OR concept_code LIKE 'C09%')
)
SELECT 
  ac.drug_class,
  COUNT(DISTINCT de.person_id) as unique_patients,
  COUNT(de.drug_exposure_id) as total_prescriptions,
  ROUND((COUNT(DISTINCT de.person_id) * 100.0 / tp.total_count), 2) as percentage_of_age_group
FROM omop.drug_exposure de
INNER JOIN age_filter af ON de.person_id = af.person_id
INNER JOIN omop.concept_ancestor ca ON de.drug_concept_id = ca.descendant_concept_id
INNER JOIN antihypertensive_classes ac ON ca.ancestor_concept_id = ac.concept_id
GROUP BY ac.drug_class
ORDER BY unique_patients DESC;
```
