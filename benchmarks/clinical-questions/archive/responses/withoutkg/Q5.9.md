## **SQL Query Used:**
```sql
WITH ckd_patients AS (
    SELECT DISTINCT co.person_id
    FROM omop.condition_occurrence co
    WHERE co.condition_concept_id IN (
        443612, 3559378, 3553780, 3553408, 3577570, -- CKD Stage 4
        443611, 3553409, 3559379, 3577571, 3577573 -- CKD Stage 5 (core concepts only)
    )
),
outpatient_labs AS (
    SELECT 
        m.person_id,
        m.measurement_concept_id,
        c.concept_name as lab_name,
        COUNT(*) as order_count
    FROM omop.measurement m
    INNER JOIN ckd_patients ckd ON m.person_id = ckd.person_id
    INNER JOIN omop.concept c ON m.measurement_concept_id = c.concept_id
    INNER JOIN omop.visit_occurrence vo ON m.visit_occurrence_id = vo.visit_occurrence_id
    WHERE vo.visit_concept_id IN (9202, 9201, 581477) -- outpatient visits
    AND m.measurement_date >= '2020-01-01'  -- recent data
    GROUP BY m.person_id, m.measurement_concept_id, c.concept_name
)
SELECT 
    lab_name,
    measurement_concept_id,
    COUNT(DISTINCT person_id) as patient_count,
    SUM(order_count) as total_orders,
    ROUND(AVG(CAST(order_count AS FLOAT)), 2) as avg_orders_per_patient
FROM outpatient_labs
GROUP BY lab_name, measurement_concept_id
HAVING COUNT(DISTINCT person_id) >= 10  -- at least 10 patients
ORDER BY COUNT(DISTINCT person_id) DESC, SUM(order_count) DESC
```
