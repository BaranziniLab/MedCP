WITH pe_encounters AS (

),
ddimer_within_7d AS (
  SELECT DISTINCT pe.visit_occurrence_id, pe.condition_concept_id, 1 as had_ddimer,
    MAX(CASE 
      WHEN m.value_as_number IS NOT NULL AND m.value_as_number > 500 THEN 1
      WHEN LOWER(m.value_source_value) LIKE '%positive%' OR LOWER(m.value_source_value) LIKE '%elevated%' OR LOWER(m.value_source_value) LIKE '%high%' THEN 1
