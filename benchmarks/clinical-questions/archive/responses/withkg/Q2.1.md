WITH cv_conditions AS (
  SELECT DISTINCT concept_id
  FROM omop.concept 
  WHERE domain_id = 'Condition' 
  AND standard_concept = 'S'
  AND (concept_id IN (320128, 316866, 316139, 319835, 313217, 312327, 315286, 321588, 321052, 381591, 434376, 436706, 438170, 438438, 438447, 441579, 4329847, 4108832)
       OR concept_name LIKE 'Hypertensive%'
       OR concept_name LIKE 'Heart failure%'
       OR concept_name LIKE '%myocardial infarction%'
       OR concept_name LIKE 'Coronary%'
       OR concept_name LIKE 'Atrial fibrillation%'
       OR concept_name LIKE 'Congestive heart failure%'
       OR concept_name LIKE 'Stroke%'
       OR concept_name LIKE 'Cerebrovascular%'
       OR concept_name LIKE 'Peripheral vascular%'
       OR concept_name LIKE 'Atherosclerosis%'
       OR concept_name LIKE '%ischemic heart%')
),
patient_cv_count AS (
  SELECT 
    p.person_id,
    p.gender_concept_id,
    COUNT(DISTINCT co.condition_concept_id) as cv_condition_count
  FROM omop.person p
  INNER JOIN omop.condition_occurrence co ON p.person_id = co.person_id
  INNER JOIN cv_conditions cv ON co.condition_concept_id = cv.concept_id
  WHERE p.gender_concept_id IN (8507, 8532)
  GROUP BY p.person_id, p.gender_concept_id
)
SELECT 
  CASE WHEN gender_concept_id = 8507 THEN 'Male' ELSE 'Female' END as gender,
  COUNT(*) as total_cv_patients,
  SUM(CASE WHEN cv_condition_count >= 2 THEN 1 ELSE 0 END) as multimorbidity_patients,
  CAST(SUM(CASE WHEN cv_condition_count >= 2 THEN 1 ELSE 0 END) * 100.0 / COUNT(*) AS DECIMAL(5,2)) as multimorbidity_percentage,
  AVG(CAST(cv_condition_count AS FLOAT)) as avg_cv_conditions
FROM patient_cv_count
WHERE cv_condition_count > 0
GROUP BY gender_concept_id
ORDER BY gender_concept_id
```
