## Analysis: Most Frequent New Diagnoses Within 180 Days Post-Severe COVID-19 Hospitalization

### **SQL Query Used:**
```sql
WITH covid_hospitalizations AS (
    SELECT DISTINCT 
        co.person_id,
        vo.visit_start_date as covid_hospitalization_date
    FROM omop.condition_occurrence co
    JOIN omop.visit_occurrence vo ON co.visit_occurrence_id = vo.visit_occurrence_id
    JOIN omop.concept c ON co.condition_concept_id = c.concept_id
    WHERE c.concept_id IN (37311061, 3661408, 605554, 3661406, 37310284, 3655976)
       AND vo.visit_concept_id = 9201
),
baseline_conditions AS (
    SELECT DISTINCT 
        ch.person_id,
        co.condition_concept_id
    FROM covid_hospitalizations ch
    JOIN omop.condition_occurrence co ON ch.person_id = co.person_id
    WHERE co.condition_start_date < ch.covid_hospitalization_date
),
post_covid_new_diagnoses AS (
    SELECT 
        ch.person_id,
        co.condition_concept_id,
        c.concept_name as condition_name,
        co.condition_start_date,
        ch.covid_hospitalization_date
    FROM covid_hospitalizations ch
    JOIN omop.condition_occurrence co ON ch.person_id = co.person_id
    JOIN omop.concept c ON co.condition_concept_id = c.concept_id
    LEFT JOIN baseline_conditions bc ON ch.person_id = bc.person_id 
                                    AND co.condition_concept_id = bc.condition_concept_id
    WHERE co.condition_start_date > ch.covid_hospitalization_date
      AND co.condition_start_date <= DATEADD(day, 180, ch.covid_hospitalization_date)
      AND bc.condition_concept_id IS NULL
      AND c.concept_id != 0
      AND c.invalid_reason IS NULL
)
SELECT TOP 20
    condition_concept_id,
    condition_name,
    COUNT(DISTINCT person_id) as patient_count,
    COUNT(*) as total_occurrences,
    CAST(100.0 * COUNT(DISTINCT person_id) / 5117.0 AS DECIMAL(5,2)) as percentage_of_covid_patients
FROM post_covid_new_diagnoses
GROUP BY condition_concept_id, condition_name
HAVING COUNT(DISTINCT person_id) >= 10
ORDER BY COUNT(DISTINCT person_id) DESC;
```
