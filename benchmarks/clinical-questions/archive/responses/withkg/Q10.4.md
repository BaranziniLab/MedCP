FROM (

          THEN 'Highly_Beta1_Selective'
          WHEN LOWER(c.concept_name) LIKE '%propranolol%' OR LOWER(c.concept_name) LIKE '%carvedilol%' 
               OR LOWER(c.concept_name) LIKE '%labetalol%'
          THEN 'Less_Selective'
          ELSE 'Other'
        END as beta_blocker_class,
        CASE WHEN readmit.days_to_readmit <= 90 AND readmit.days_to_readmit IS NOT NULL THEN 1 ELSE 0 END as readmit_90day
    FROM (
        SELECT DISTINCT co.person_id, co.condition_start_date as mi_date
        FROM omop.condition_occurrence co
        INNER JOIN omop.concept con ON co.condition_concept_id = con.concept_id
        WHERE LOWER(con.concept_name) LIKE '%myocardial infarction%'

    INNER JOIN (
        SELECT m.person_id
        FROM omop.measurement m
        INNER JOIN omop.concept mc ON m.measurement_concept_id = mc.concept_id
