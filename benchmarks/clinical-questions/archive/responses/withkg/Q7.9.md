SELECT TOP 20
    c.concept_name as drug_name,
    COUNT(DISTINCT de.person_id) as patient_count,
    COUNT(de.drug_exposure_id) as prescription_count
FROM omop.drug_exposure de
JOIN omop.concept c ON de.drug_concept_id = c.concept_id
JOIN omop.condition_occurrence co ON de.person_id = co.person_id
JOIN omop.concept c2 ON co.condition_concept_id = c2.concept_id
WHERE (LOWER(c2.concept_name) LIKE '%attention deficit%'
       OR LOWER(c2.concept_name) LIKE '%adhd%'
       OR LOWER(c2.concept_name) LIKE '%hyperactivity%')
   AND (LOWER(c.concept_name) LIKE '%methylphenidate%' 
        OR LOWER(c.concept_name) LIKE '%amphetamine%' 
        OR LOWER(c.concept_name) LIKE '%atomoxetine%'
        OR LOWER(c.concept_name) LIKE '%lisdexamfetamine%'
        OR LOWER(c.concept_name) LIKE '%dextroamphetamine%')
   AND ABS(DATEDIFF(day, de.drug_exposure_start_date, co.condition_start_date)) <= 365
GROUP BY c.concept_name
ORDER BY patient_count DESC
```
