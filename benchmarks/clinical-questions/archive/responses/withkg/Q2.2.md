```sql
-- Finding metformin concepts
SELECT TOP 20 c.concept_name, c.concept_id, c.concept_class_id
FROM omop.concept c
WHERE LOWER(c.concept_name) LIKE '%metformin%' 
    AND c.domain_id = 'Drug' AND c.standard_concept = 'S'
ORDER BY c.concept_name;

-- Counting patients by formulation
SELECT TOP 10 c.concept_name, c.concept_id, 
    COUNT(DISTINCT de.person_id) as patient_count,
    COUNT(*) as total_exposures
FROM omop.drug_exposure de
JOIN omop.concept c ON de.drug_concept_id = c.concept_id
WHERE LOWER(c.concept_name) LIKE '%metformin%' 
    AND c.domain_id = 'Drug' AND c.standard_concept = 'S'
GROUP BY c.concept_name, c.concept_id
ORDER BY patient_count DESC;

-- Total unique metformin users
SELECT COUNT(DISTINCT de.person_id) as total_unique_patients_on_metformin
FROM omop.drug_exposure de
JOIN omop.concept c ON de.drug_concept_id = c.concept_id
WHERE LOWER(c.concept_name) LIKE '%metformin%' 
    AND c.domain_id = 'Drug' AND c.standard_concept = 'S';
```
