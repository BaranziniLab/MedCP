WITH type2_diabetes_concepts AS (
    SELECT concept_id
    FROM omop.concept
    WHERE standard_concept = 'S'
      AND (LOWER(concept_name) LIKE '%type 2 diabetes%' 
           OR LOWER(concept_name) LIKE '%diabetes mellitus type 2%'
           OR concept_id = 201826
           OR concept_id = 4193704)
)
SELECT COUNT(DISTINCT person_id) as total_type2_diabetes_patients
FROM omop.condition_occurrence co
WHERE co.condition_concept_id IN (SELECT concept_id FROM type2_diabetes_concepts);
