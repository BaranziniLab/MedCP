### SQL Query Used:
```sql
WITH women_in_50s AS (
    SELECT person_id FROM omop.person 
    WHERE gender_concept_id = 8532 AND (2023 - year_of_birth) BETWEEN 50 AND 59
),
diagnosis_counts AS (
    SELECT co.condition_concept_id, c.concept_name as diagnosis_name,
           COUNT(*) as frequency, COUNT(DISTINCT co.person_id) as unique_patients
    FROM omop.condition_occurrence co
    INNER JOIN women_in_50s w ON co.person_id = w.person_id
    INNER JOIN omop.concept c ON co.condition_concept_id = c.concept_id
    WHERE c.domain_id = 'Condition' AND c.standard_concept = 'S'
    GROUP BY co.condition_concept_id, c.concept_name
)
SELECT TOP 15 diagnosis_name, frequency, unique_patients,
       percentage_of_cohort FROM diagnosis_counts ORDER BY frequency DESC
```
