-- Main analysis query to identify new SSRI users and calculate 6-month discontinuation rates
WITH ssri_ingredients AS (
    SELECT concept_id FROM (VALUES 
        (715939), -- escitalopram
        (722031), -- paroxetine  
        (739138), -- sertraline
        (751412), -- fluvoxamine
        (755695), -- fluoxetine
        (797617)  -- citalopram
    ) AS t(concept_id)
),
ssri_concepts AS (
    SELECT DISTINCT ca.descendant_concept_id as concept_id
    FROM omop.concept_ancestor ca
    INNER JOIN ssri_ingredients si ON ca.ancestor_concept_id = si.concept_id
),
new_ssri_users AS (
    SELECT 
        de.person_id,
        MIN(de.drug_era_start_date) as first_ssri_date,
        de.drug_concept_id,
        de.drug_era_start_date,
        de.drug_era_end_date,
        DATEDIFF(day, de.drug_era_start_date, de.drug_era_end_date) + 1 as era_duration_days
    FROM omop.drug_era de
    INNER JOIN ssri_concepts sc ON de.drug_concept_id = sc.concept_id
    GROUP BY de.person_id, de.drug_concept_id, de.drug_era_start_date, de.drug_era_end_date
),
first_ssri_era AS (
    SELECT 
        person_id,
        first_ssri_date,
        drug_concept_id,
        drug_era_start_date,
        drug_era_end_date,
        era_duration_days,
        ROW_NUMBER() OVER (PARTITION BY person_id ORDER BY first_ssri_date) as rn
    FROM new_ssri_users
    WHERE drug_era_start_date = first_ssri_date
)
SELECT 
    COUNT(*) as total_new_ssri_users,
    SUM(CASE WHEN era_duration_days < 180 THEN 1 ELSE 0 END) as discontinued_within_6_months,
    ROUND(100.0 * SUM(CASE WHEN era_duration_days < 180 THEN 1 ELSE 0 END) / COUNT(*), 2) as discontinuation_rate_6_months_pct,
    AVG(CAST(era_duration_days as FLOAT)) as avg_era_duration_days
FROM first_ssri_era
WHERE rn = 1
```
